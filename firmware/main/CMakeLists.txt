idf_component_register(SRCS "main.c"
                    INCLUDE_DIRS "." "${CMAKE_CURRENT_BINARY_DIR}"
                    REQUIRES esp_wifi nvs_flash esp_netif esp_event esp_websocket_client driver esp_driver_i2s esp_driver_gpio esp_driver_rmt)

# Look for config.env in project root or firmware dir
set(CONFIG_ENV_PATH "")
if(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../../config.env")
    set(CONFIG_ENV_PATH "${CMAKE_CURRENT_LIST_DIR}/../../config.env")
elseif(EXISTS "${CMAKE_CURRENT_LIST_DIR}/../config.env")
    set(CONFIG_ENV_PATH "${CMAKE_CURRENT_LIST_DIR}/../config.env")
endif()

if(CONFIG_ENV_PATH)
    message(STATUS "[Config] Found unified config.env: ${CONFIG_ENV_PATH}")
    file(STRINGS "${CONFIG_ENV_PATH}" CONFIG_LINES ENCODING UTF-8)

    set(CFG_WIFI_SSID "")
    set(CFG_WIFI_PASSWORD "")
    set(CFG_SERVER_HOST "")
    set(CFG_SERVER_PORT "8000")
    set(CFG_I2S_MIC_WS_GPIO "4")
    set(CFG_I2S_MIC_SCK_GPIO "5")
    set(CFG_I2S_MIC_SD_GPIO "6")
    set(CFG_AUDIO_GAIN_BOOST "2")

    foreach(line ${CONFIG_LINES})
        string(STRIP "${line}" line)
        if(NOT line MATCHES "^#" AND line MATCHES "^([A-Za-z0-9_]+)=(.*)$")
            set(KEY "${CMAKE_MATCH_1}")
            set(VAL "${CMAKE_MATCH_2}")
            string(REGEX REPLACE "^[\"'](.*)[\"']$" "\\1" VAL "${VAL}")
            if(KEY STREQUAL "WIFI_SSID")
                set(CFG_WIFI_SSID "${VAL}")
            elseif(KEY STREQUAL "WIFI_PASSWORD")
                set(CFG_WIFI_PASSWORD "${VAL}")
            elseif(KEY STREQUAL "SERVER_HOST" OR KEY STREQUAL "SERVER_IP")
                set(CFG_SERVER_HOST "${VAL}")
            elseif(KEY STREQUAL "SERVER_PORT")
                set(CFG_SERVER_PORT "${VAL}")
            elseif(KEY STREQUAL "I2S_MIC_WS_GPIO")
                set(CFG_I2S_MIC_WS_GPIO "${VAL}")
            elseif(KEY STREQUAL "I2S_MIC_SCK_GPIO")
                set(CFG_I2S_MIC_SCK_GPIO "${VAL}")
            elseif(KEY STREQUAL "I2S_MIC_SD_GPIO")
                set(CFG_I2S_MIC_SD_GPIO "${VAL}")
            elseif(KEY STREQUAL "AUDIO_GAIN_BOOST")
                set(CFG_AUDIO_GAIN_BOOST "${VAL}")
            endif()
        endif()
    endforeach()

    set(HEADER_CONTENT "// Auto-generated by CMake from config.env. Do NOT edit manually.\n#pragma once\n\n")
    if(NOT CFG_WIFI_SSID STREQUAL "")
        string(APPEND HEADER_CONTENT "#define APP_CONFIG_WIFI_SSID \"${CFG_WIFI_SSID}\"\n")
    endif()
    if(NOT CFG_WIFI_PASSWORD STREQUAL "")
        string(APPEND HEADER_CONTENT "#define APP_CONFIG_WIFI_PASSWORD \"${CFG_WIFI_PASSWORD}\"\n")
    endif()
    if(NOT CFG_SERVER_HOST STREQUAL "")
        string(APPEND HEADER_CONTENT "#define APP_CONFIG_SERVER_HOST \"${CFG_SERVER_HOST}\"\n")
    endif()
    if(NOT CFG_SERVER_PORT STREQUAL "")
        string(APPEND HEADER_CONTENT "#define APP_CONFIG_SERVER_PORT ${CFG_SERVER_PORT}\n")
    endif()
    if(NOT CFG_I2S_MIC_WS_GPIO STREQUAL "")
        string(APPEND HEADER_CONTENT "#define APP_CONFIG_I2S_MIC_WS_GPIO ${CFG_I2S_MIC_WS_GPIO}\n")
    endif()
    if(NOT CFG_I2S_MIC_SCK_GPIO STREQUAL "")
        string(APPEND HEADER_CONTENT "#define APP_CONFIG_I2S_MIC_SCK_GPIO ${CFG_I2S_MIC_SCK_GPIO}\n")
    endif()
    if(NOT CFG_I2S_MIC_SD_GPIO STREQUAL "")
        string(APPEND HEADER_CONTENT "#define APP_CONFIG_I2S_MIC_SD_GPIO ${CFG_I2S_MIC_SD_GPIO}\n")
    endif()
    if(NOT CFG_AUDIO_GAIN_BOOST STREQUAL "")
        string(APPEND HEADER_CONTENT "#define APP_CONFIG_AUDIO_GAIN_BOOST ${CFG_AUDIO_GAIN_BOOST}\n")
    endif()

    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/app_config_auto.h" "${HEADER_CONTENT}")
    message(STATUS "[Config] Generated app_config_auto.h with WiFi and Server Host from config.env")
endif()
